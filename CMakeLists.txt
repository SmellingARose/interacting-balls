cmake_minimum_required(VERSION 3.16)
project(brtrain C)

option(BR_METAL  "Metal backend (Apple GPUs)" ${APPLE})
option(BR_OPENCL "OpenCL backend (AMD, NVIDIA and Intel GPUs)" ON)
option(BR_PORTABLE "Build for other computers: not tuned to this computer's processor (see BR_ARCH)" OFF)
set(BR_ARCH "" CACHE STRING "Processor level to build for instead of this computer's: e.g. x86-64-v2, x86-64-v3 (GCC, Clang) or AVX2 (MSVC)")
set(CMAKE_C_STANDARD 11)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

set(EMBED ${CMAKE_BINARY_DIR}/kernels_embed.c)
file(GLOB_RECURSE VENDOR_FILES ${CMAKE_SOURCE_DIR}/web/vendor/*.js)
file(GLOB KERNEL_FILES ${CMAKE_SOURCE_DIR}/src/kernels/* ${CMAKE_SOURCE_DIR}/web/index.html ${VENDOR_FILES})
add_custom_command(OUTPUT ${EMBED}
  COMMAND ${CMAKE_COMMAND} -DOUT=${EMBED} -DSRC=${CMAKE_SOURCE_DIR}/src -P ${CMAKE_SOURCE_DIR}/cmake/embed.cmake
  DEPENDS ${CMAKE_SOURCE_DIR}/src/sim_core.h ${KERNEL_FILES} ${CMAKE_SOURCE_DIR}/cmake/embed.cmake)

add_executable(brtrain src/main.c src/trainer.c src/server.c src/backend_cpu.c ${EMBED})
target_include_directories(brtrain PRIVATE src)

# CPU code: tuned to the processor of the computer it is built on (GCC/Clang: -march=native, or -mcpu=native on Arm,
# which also sets the tuning; MSVC: AVX2 when this processor has it). A program meant for other computers is built
# with BR_PORTABLE (the compiler's baseline: runs on any processor of the architecture) or a chosen BR_ARCH level.
# Cross and multi-architecture builds are never tuned to the build machine.
include(CheckCCompilerFlag)
set(BR_PROC ${CMAKE_SYSTEM_PROCESSOR})
if(CMAKE_OSX_ARCHITECTURES)
  string(REPLACE ";" "/" BR_PROC "${CMAKE_OSX_ARCHITECTURES}")   # several (a universal build): never the build machine's
endif()
set(BR_NATIVE ON)
set(BR_PROC_NAME "${BR_PROC}")
if(NOT BR_PROC_NAME)   # a cross build that names no processor (only CMAKE_SYSTEM_NAME): the compiler's target says which
  if(CMAKE_C_COMPILER_TARGET)
    set(BR_PROC_NAME "${CMAKE_C_COMPILER_TARGET}")
  elseif(NOT MSVC)
    execute_process(COMMAND ${CMAKE_C_COMPILER} -dumpmachine OUTPUT_VARIABLE BR_PROC_NAME OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  endif()
  string(REGEX REPLACE "-.*" "" BR_PROC_NAME "${BR_PROC_NAME}")   # x86_64-w64-mingw32 -> x86_64
endif()
if(BR_PROC_NAME)
  set(BR_CPU_FOR "any ${BR_PROC_NAME} processor")
else()
  set(BR_CPU_FOR "the compiler's baseline processor")
endif()
if(BR_PORTABLE OR BR_ARCH OR CMAKE_CROSSCOMPILING OR NOT BR_PROC STREQUAL CMAKE_HOST_SYSTEM_PROCESSOR)
  set(BR_NATIVE OFF)
endif()
if(MSVC)
  target_compile_options(brtrain PRIVATE /O2 /fp:fast)
  target_compile_definitions(brtrain PRIVATE _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
  if(BR_ARCH)
    target_compile_options(brtrain PRIVATE /arch:${BR_ARCH})
    set(BR_CPU_FOR "${BR_ARCH}")
  elseif(BR_NATIVE)
    include(CheckCSourceRuns)
    check_c_source_runs("#include <intrin.h>
int main(void) { int r[4]; __cpuid(r, 0); if (r[0] < 7) return 1; __cpuid(r, 1);
  if (!(r[2] & (1 << 12)) || !(r[2] & (1 << 27)) || !(r[2] & (1 << 28)) || (_xgetbv(0) & 6) != 6) return 1;
  __cpuidex(r, 7, 0); return (r[1] & (1 << 5)) ? 0 : 1; }" BR_HOST_AVX2)
    if(BR_HOST_AVX2)
      target_compile_options(brtrain PRIVATE /arch:AVX2)
      set(BR_CPU_FOR "this computer's processor (AVX2)")
    endif()
  endif()
else()
  target_compile_options(brtrain PRIVATE -O3 -ffast-math)
  if(BR_ARCH)
    target_compile_options(brtrain PRIVATE -march=${BR_ARCH})
    set(BR_CPU_FOR "${BR_ARCH}")
  elseif(BR_NATIVE AND BR_PROC MATCHES "^(arm64|ARM64|aarch64|AARCH64)$")
    check_c_compiler_flag(-mcpu=native HAS_MCPU_NATIVE)
    if(HAS_MCPU_NATIVE)
      target_compile_options(brtrain PRIVATE -mcpu=native)
      set(BR_CPU_FOR "this computer's processor")
    endif()
  elseif(BR_NATIVE)
    check_c_compiler_flag(-march=native HAS_MARCH_NATIVE)   # (on x86 -mcpu only sets the tuning: never a substitute)
    if(HAS_MARCH_NATIVE)
      target_compile_options(brtrain PRIVATE -march=native)
      set(BR_CPU_FOR "this computer's processor")
    endif()
  endif()
endif()
message(STATUS "brtrain: CPU code for ${BR_CPU_FOR}")

if(WIN32)
  # threads.h uses Win32 threads; a MinGW build links its runtime statically, so the program needs no MinGW DLLs
  target_link_libraries(brtrain PRIVATE ws2_32 shell32)
  if(MINGW)
    target_link_options(brtrain PRIVATE -static)
  endif()
else()
  find_package(Threads REQUIRED)
  target_link_libraries(brtrain PRIVATE Threads::Threads m)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(brtrain PRIVATE _GNU_SOURCE)   # sched_getaffinity / CPU_COUNT (threads.h)
  endif()
endif()

if(BR_METAL AND APPLE)
  enable_language(OBJC)
  target_sources(brtrain PRIVATE src/backend_metal.m)
  set_source_files_properties(src/backend_metal.m PROPERTIES COMPILE_FLAGS "-fobjc-arc")
  target_link_libraries(brtrain PRIVATE "-framework Metal" "-framework Foundation")
  target_compile_definitions(brtrain PRIVATE BR_HAVE_METAL)
  message(STATUS "brtrain: Metal backend ON")
endif()

if(BR_OPENCL)
  # OpenCL is loaded at runtime from the GPU driver (src/br_cl.h): nothing to find or link at build time
  target_sources(brtrain PRIVATE src/backend_opencl.c)
  target_compile_definitions(brtrain PRIVATE BR_HAVE_OPENCL)
  if(NOT WIN32)
    target_link_libraries(brtrain PRIVATE ${CMAKE_DL_LIBS})
  endif()
endif()
